using System;
using System.Collections.Generic;
using System.Drawing;
using System.Drawing.Drawing2D;
using System.Drawing.Imaging;
using System.IO;

// ClipStack ikonu: arkada üst üste kaydırılmış iki kâğıt, önde kıskaçlı bir pano.
// 256 birimlik tuvalde tasarlanır; küçük boyutlarda ayrıntılar sadeleşir.
string outDir = args[0];
Directory.CreateDirectory(outDir);
int[] sizes = { 16, 20, 24, 32, 40, 48, 64, 256 };
var pngs = new List<(int size, byte[] data)>();
foreach (int s in sizes)
{
    using var bmp = Render(s);
    using var ms = new MemoryStream();
    bmp.Save(ms, ImageFormat.Png);
    pngs.Add((s, ms.ToArray()));
    if (s is 16 or 32 or 48 or 256) bmp.Save(Path.Combine(outDir, $"preview-{s}.png"), ImageFormat.Png);
}
WriteIco(Path.Combine(outDir, "ClipStack.ico"), pngs);

// Önizleme: tüm boyutlar açık ve koyu zemin üzerinde, büyütülmüş.
using (var sheet = new Bitmap(900, 360))
using (var g = Graphics.FromImage(sheet))
{
    g.InterpolationMode = InterpolationMode.NearestNeighbor;
    g.PixelOffsetMode = PixelOffsetMode.Half;
    g.FillRectangle(new SolidBrush(Color.FromArgb(243, 244, 246)), 0, 0, 900, 180);
    g.FillRectangle(new SolidBrush(Color.FromArgb(31, 33, 40)), 0, 180, 900, 180);
    int x = 10;
    foreach (int s in new[] { 16, 24, 32, 48, 64 })
    {
        using var b = Render(s);
        int z = s <= 32 ? 4 : 2;
        foreach (int y0 in new[] { 0, 180 })
            g.DrawImage(b, new Rectangle(x, y0 + (180 - s * z) / 2, s * z, s * z));
        x += s * z + 16;
    }
    using var big = Render(256);
    g.InterpolationMode = InterpolationMode.HighQualityBicubic;
    g.DrawImage(big, new Rectangle(x, 10, 160, 160));
    g.DrawImage(big, new Rectangle(x, 190, 160, 160));
    sheet.Save(Path.Combine(outDir, "preview-sheet.png"), ImageFormat.Png);
}
Console.WriteLine("ok");

static Bitmap Render(int size)
{
    var bmp = new Bitmap(size, size, PixelFormat.Format32bppArgb);
    using var g = Graphics.FromImage(bmp);
    g.SmoothingMode = SmoothingMode.AntiAlias;
    g.PixelOffsetMode = PixelOffsetMode.HighQuality;
    g.CompositingQuality = CompositingQuality.HighQuality;
    float k = size / 256f;
    g.ScaleTransform(k, k);
    bool tiny = size <= 24;   // 16–24 px: iki kâğıt yerine bir, tek kalın satır
    bool small = size <= 48;

    Color back2 = Color.FromArgb(0xA9, 0xC1, 0xF7);
    Color back1 = Color.FromArgb(0x6E, 0x96, 0xEE);
    Color boardTop = Color.FromArgb(0x4A, 0x7C, 0xF0);
    Color boardBottom = Color.FromArgb(0x2B, 0x55, 0xC8);
    Color clip = Color.FromArgb(0x24, 0x33, 0x5E);
    Color line = Color.FromArgb(0xB9, 0xCB, 0xF3);

    // Arkadaki kâğıtlar (yığın)
    if (!tiny) Fill(g, new RectangleF(84, 14, 158, 180), 24, back2);
    Fill(g, tiny ? new RectangleF(76, 22, 166, 186) : new RectangleF(58, 38, 158, 180), 24, back1);

    // Önde pano gövdesi
    var board = new RectangleF(14, 62, 172, 188);
    using (var path = Rounded(board, 28))
    using (var brush = new LinearGradientBrush(board, boardTop, boardBottom, 90f))
        g.FillPath(brush, path);

    // Kâğıt
    var paper = tiny ? new RectangleF(40, 104, 120, 124) : new RectangleF(34, 98, 132, 134);
    Fill(g, paper, tiny ? 10 : 14, Color.White);

    // Satırlar
    if (tiny)
    {
        Fill(g, new RectangleF(58, 138, 84, 26), 8, line);
        Fill(g, new RectangleF(58, 182, 56, 26), 8, line);
    }
    else
    {
        float h = small ? 16 : 12;
        Fill(g, new RectangleF(54, 130, 92, h), h / 2, line);
        Fill(g, new RectangleF(54, 160, 92, h), h / 2, line);
        Fill(g, new RectangleF(54, 190, 58, h), h / 2, line);
    }

    // Kıskaç
    var clipRect = tiny ? new RectangleF(60, 44, 80, 48) : new RectangleF(64, 44, 72, 42);
    Fill(g, clipRect, tiny ? 12 : 14, clip);
    if (!tiny)
        Fill(g, new RectangleF(88, 54, 24, 12), 6, boardTop); // kıskaç deliği
    return bmp;
}

static void Fill(Graphics g, RectangleF r, float radius, Color c)
{
    using var path = Rounded(r, radius);
    using var b = new SolidBrush(c);
    g.FillPath(b, path);
}

static GraphicsPath Rounded(RectangleF r, float radius)
{
    float d = Math.Min(radius * 2, Math.Min(r.Width, r.Height));
    var p = new GraphicsPath();
    p.AddArc(r.X, r.Y, d, d, 180, 90);
    p.AddArc(r.Right - d, r.Y, d, d, 270, 90);
    p.AddArc(r.Right - d, r.Bottom - d, d, d, 0, 90);
    p.AddArc(r.X, r.Bottom - d, d, d, 90, 90);
    p.CloseFigure();
    return p;
}

// PNG gömülü ICO (Windows Vista+ tüm boyutlarda destekler).
static void WriteIco(string path, List<(int size, byte[] data)> images)
{
    using var fs = File.Create(path);
    using var w = new BinaryWriter(fs);
    w.Write((short)0); w.Write((short)1); w.Write((short)images.Count);
    int offset = 6 + 16 * images.Count;
    foreach (var (size, data) in images)
    {
        w.Write((byte)(size >= 256 ? 0 : size));
        w.Write((byte)(size >= 256 ? 0 : size));
        w.Write((byte)0); w.Write((byte)0);
        w.Write((short)1); w.Write((short)32);
        w.Write(data.Length); w.Write(offset);
        offset += data.Length;
    }
    foreach (var (_, data) in images) w.Write(data);
}
